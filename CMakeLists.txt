cmake_minimum_required(VERSION 3.24)
project(ShareKeys VERSION 0.1.0 LANGUAGES CXX)
if(WIN32)
  enable_language(RC)
  # Match the installer and executable destinations. Qt otherwise deploys to bin/.
  set(CMAKE_INSTALL_BINDIR ".")
  set(CMAKE_INSTALL_LIBDIR ".")
endif()
include(GNUInstallDirs)
configure_file(cmake/build_info.hpp.in generated/sharekeys/build_info.hpp @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
add_library(sharekeys_core src/core/keyboard.cpp src/core/mouse.cpp src/protocol/protocol.cpp src/routing/router.cpp src/routing/workspace.cpp)
target_include_directories(sharekeys_core PUBLIC include)
target_link_libraries(sharekeys_core PUBLIC Threads::Threads)
function(sharekeys_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
    target_compile_definitions(${target} PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endfunction()
sharekeys_warnings(sharekeys_core)
find_package(OpenSSL REQUIRED)
find_path(ASIO_INCLUDE_DIR asio.hpp)
if(NOT ASIO_INCLUDE_DIR)
  include(FetchContent)
  FetchContent_Declare(asio
    URL https://github.com/chriskohlhoff/asio/archive/refs/tags/asio-1-32-0.tar.gz
    URL_HASH SHA256=f1b94b80eeb00bb63a3c8cef5047d4e409df4d8a3fe502305976965827d95672)
  FetchContent_MakeAvailable(asio)
  set(ASIO_INCLUDE_DIR "${asio_SOURCE_DIR}/asio/include")
endif()
add_library(sharekeys_network src/network/network.cpp src/security/security.cpp src/core/host_engine.cpp src/core/agent_service.cpp)
target_include_directories(sharekeys_network SYSTEM PUBLIC "${ASIO_INCLUDE_DIR}")
target_compile_definitions(sharekeys_network PUBLIC ASIO_STANDALONE)
target_link_libraries(sharekeys_network PUBLIC sharekeys_core OpenSSL::SSL OpenSSL::Crypto)
sharekeys_warnings(sharekeys_network)
if(WIN32)
  target_link_libraries(sharekeys_network PUBLIC ws2_32 mswsock crypt32)
endif()
if(WIN32)
  add_library(sharekeys_input src/input/win32_keyboard.cpp src/input/win32_mouse.cpp)
  target_link_libraries(sharekeys_input PUBLIC sharekeys_core user32)
  sharekeys_warnings(sharekeys_input)
endif()
option(SHAREKEYS_BUILD_GUI "Build Qt host (Linux builds are development previews)" ON)
if(SHAREKEYS_BUILD_GUI OR WIN32)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Network)
  add_library(sharekeys_desktop src/gui/desktop_settings.cpp)
  target_link_libraries(sharekeys_desktop PUBLIC sharekeys_network Qt6::Widgets Qt6::Network)
  sharekeys_warnings(sharekeys_desktop)
  add_library(sharekeys_desktop_ui src/gui/host_controller.cpp include/sharekeys/host_controller.hpp
    src/gui/host_window.cpp src/gui/agent_window.cpp src/gui/desktop_application.cpp resources/desktop.qrc)
  set_target_properties(sharekeys_desktop_ui PROPERTIES AUTOMOC ON AUTORCC ON)
  target_include_directories(sharekeys_desktop_ui PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(sharekeys_desktop_ui PRIVATE SHAREKEYS_BUILD_CONFIG="$<CONFIG>")
  target_link_libraries(sharekeys_desktop_ui PUBLIC sharekeys_desktop)
  sharekeys_warnings(sharekeys_desktop_ui)
endif()
if(SHAREKEYS_BUILD_GUI)
  qt_add_executable(sharekeys_host apps/host/main.cpp)
  target_link_libraries(sharekeys_host PRIVATE sharekeys_desktop_ui)
  sharekeys_warnings(sharekeys_host)
  if(WIN32)
    target_link_libraries(sharekeys_host PRIVATE sharekeys_input)
  endif()
endif()
if(SHAREKEYS_BUILD_GUI OR WIN32)
  qt_add_executable(sharekeys_agent apps/agent/gui_main.cpp)
  target_link_libraries(sharekeys_agent PRIVATE sharekeys_desktop_ui)
  sharekeys_warnings(sharekeys_agent)
endif()
if(WIN32)
  foreach(role host agent)
    if(TARGET sharekeys_${role})
      set(SHAREKEYS_RESOURCE_ROLE "${role}")
      set(SHAREKEYS_RESOURCE_TARGET "${role}")
      configure_file(resources/windows/application.rc.in generated/${role}.rc @ONLY)
      target_sources(sharekeys_${role} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/${role}.rc")
    endif()
  endforeach()
  set_target_properties(sharekeys_agent PROPERTIES WIN32_EXECUTABLE TRUE)
  if(SHAREKEYS_BUILD_GUI)
    set_target_properties(sharekeys_host PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  target_link_libraries(sharekeys_agent PRIVATE sharekeys_input)
  add_executable(sharekeys_agent_cli apps/agent/main.cpp)
  target_link_libraries(sharekeys_agent_cli PRIVATE sharekeys_input sharekeys_network Qt6::Core)
  sharekeys_warnings(sharekeys_agent_cli)
  add_executable(sharekeys_windows_input_test tests/integration/windows_input.cpp)
  target_link_libraries(sharekeys_windows_input_test PRIVATE sharekeys_input)
  sharekeys_warnings(sharekeys_windows_input_test)
  install(TARGETS sharekeys_agent sharekeys_agent_cli RUNTIME DESTINATION .)
  if(SHAREKEYS_BUILD_GUI)
    install(TARGETS sharekeys_host RUNTIME DESTINATION .)
    qt_generate_deploy_app_script(TARGET sharekeys_host OUTPUT_SCRIPT host_deploy NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${host_deploy})
  endif()
  qt_generate_deploy_app_script(TARGET sharekeys_agent OUTPUT_SCRIPT agent_deploy NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${agent_deploy})
  # vcpkg app-local deployment places non-Qt DLLs (including OpenSSL) beside the
  # executable. Qt's deployment script supplies its own DLLs and plugins.
  install(DIRECTORY "$<TARGET_FILE_DIR:sharekeys_agent>/" DESTINATION . FILES_MATCHING PATTERN "*.dll" PATTERN "*gtest*.dll" EXCLUDE PATTERN "*gmock*.dll" EXCLUDE)
  # App-local MSVC runtimes keep the per-user installer independent of an
  # elevated redistributable installer on a clean Windows machine.
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ".")
  include(InstallRequiredSystemLibraries)
  install(FILES README.md docs/INSTALLATION.md docs/RELEASE.md docs/BUILD_WINDOWS.md docs/WINDOWS_TESTING.md DESTINATION docs)
  if(VCPKG_INSTALLED_DIR AND VCPKG_TARGET_TRIPLET)
    install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/"
      DESTINATION licenses FILES_MATCHING PATTERN copyright PATTERN "*license*" PATTERN "*LICENSE*")
  endif()
endif()
if(BUILD_TESTING)
  find_package(GTest CONFIG REQUIRED)
  add_executable(sharekeys_tests tests/unit/keyboard_test.cpp tests/unit/mouse_test.cpp tests/unit/protocol_test.cpp tests/unit/router_test.cpp tests/integration/network_test.cpp)
  target_link_libraries(sharekeys_tests PRIVATE sharekeys_network GTest::gtest_main)
  sharekeys_warnings(sharekeys_tests)
  include(GoogleTest)
  gtest_discover_tests(sharekeys_tests PROPERTIES TIMEOUT 20)
  if(TARGET sharekeys_desktop)
    add_executable(sharekeys_desktop_tests tests/unit/desktop_settings_test.cpp)
    target_link_libraries(sharekeys_desktop_tests PRIVATE sharekeys_desktop GTest::gtest_main)
    sharekeys_warnings(sharekeys_desktop_tests)
    gtest_discover_tests(sharekeys_desktop_tests PROPERTIES TIMEOUT 20)
    add_executable(sharekeys_gui_tests tests/integration/gui_test.cpp)
    target_link_libraries(sharekeys_gui_tests PRIVATE sharekeys_desktop_ui GTest::gtest)
    sharekeys_warnings(sharekeys_gui_tests)
    add_test(NAME Desktop.GuiIntegration COMMAND sharekeys_gui_tests)
    set_tests_properties(Desktop.GuiIntegration PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()
